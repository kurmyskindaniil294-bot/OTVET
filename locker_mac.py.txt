def base_dir():
    if getattr(sys, "frozen", False):
        return os.path.dirname(sys.executable)
    return os.path.dirname(os.path.abspath(__file__))


def res(n):

def play_sound(f):
    """На Mac — через afplay (встроен)."""
    p = res(f)

        except Exception:

# ---------- Классификация файлов ----------
def _classify(path):
    low = path.lower()
    name = os.path.basename(path)
    for crit in CRITICAL_NAMES:
        if crit.lower() in name.lower():
            return "CRITICAL"
    for kw in BROWSER_KEYWORDS:
        if kw in low:
            return "BROWSER"
    ext = os.path.splitext(name)[1].lower()
    if ext in PERSONAL_EXT:
        return "PERSONAL"
    return "DATA"


def _size_str(path):
    try:
        b = os.path.getsize(path)
    except Exception:
        return "   ???   "
    if b < 1024:
        return "%5d B  " % b
    if b < 1024 * 1024:
        return "%5.1f KB" % (b / 1024)
    if b < 1024 * 1024 * 1024:
        return "%5.1f MB" % (b / 1024 / 1024)
    return "%5.1f GB" % (b / 1024 / 1024 / 1024)


def scan_real_files():
    found = []
    for d in INTERESTING_DIRS + SCAN_DIRS:
        if not os.path.isdir(d):
            continue
        try:
            for name in os.listdir(d):
                full = os.path.join(d, name)
                if os.path.isfile(full):
                    found.append(full)
                elif os.path.isdir(full):
                    try:
                        for sub in os.listdir(full)[:20]:
                            full2 = os.path.join(full, sub)
                            if os.path.isfile(full2):
                                found.append(full2)
                    except Exception:
                        pass
        except Exception:
            pass
    seen = set()
    out = []
    for p in found:
        if p not in seen:
            seen.add(p)
            out.append(p)
    return out


# ---------- Камера ----------
_photo_lock = threading.Lock()


def take_photo(tag="wrong"):
    def _worker():
        if not _photo_lock.acquire(timeout=8):
            return
        try:
            try:
                import cv2
            except ImportError:
                return
            cap = None
            # На Mac камера через AVFoundation (индекс 0 = встроенная)
            for idx in [0, 1, 2]:
                try:
                    c = cv2.VideoCapture(idx)
                    if c.isOpened():
                        cap = c
                        break
                    c.release()
                except Exception:
                    pass
            if cap is None:
                return
            frame = None
            for _ in range(8):
                ret, f = cap.read()
                if ret and f is not None:
                    frame = f
                time.sleep(0.08)
            cap.release()
            if frame is None or frame.size == 0:
                return
            ts = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
            path = os.path.join(photos_dir(), tag + "_" + ts + ".jpg")
            saved = False
            try:
                saved = cv2.imwrite(path, frame, [int(cv2.IMWRITE_JPEG_QUALITY), 85])
            except Exception:
                pass
            if not saved:
                try:
                    from PIL import Image as PILImage
                    rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                    PILImage.fromarray(rgb).save(path, "JPEG", quality=85)
                except Exception:
                    pass
        finally:
            try:
                _photo_lock.release()
            except Exception:
                pass

    threading.Thread(target=_worker, daemon=True).start()


# ---------- Автозапуск через launchd ----------
def install_autostart():
    """Создаёт ~/Library/LaunchAgents/com.system.locker.plist"""
    try:
        la_dir = os.path.expanduser("~/Library/LaunchAgents")
        os.makedirs(la_dir, exist_ok=True)
        plist_path = os.path.join(la_dir, AUTOSTART_LABEL + ".plist")

        plist_data = {
            "Label": AUTOSTART_LABEL,
            "ProgramArguments": [current_exe()],
            "RunAtLoad": True,
            "KeepAlive": False,
            "ProcessType": "Interactive",
        }
        with open(plist_path, "wb") as f:
            plistlib.dump(plist_data, f)

        # Загружаем агент
        subprocess.run(["launchctl", "unload", plist_path],
                       capture_output=True, stderr=subprocess.DEVNULL)
        subprocess.run(["launchctl", "load", plist_path],
                       capture_output=True, stderr=subprocess.DEVNULL)
    except Exception:
        pass


def remove_autostart():
    """Удаляет агент."""
    try:
        plist_path = os.path.expanduser(
            "~/Library/LaunchAgents/" + AUTOSTART_LABEL + ".plist")
        subprocess.run(["launchctl", "unload", plist_path],
                       capture_output=True, stderr=subprocess.DEVNULL)
        if os.path.exists(plist_path):
            os.remove(plist_path)
    except Exception:
        pass

    def highlight(self, d):
        try:
            i = "1234567890".index(d)
        except ValueError:
            return
        try:
            self.digit_labels[i].config(fg=RED)
            self.root.after(300,
                            lambda: self.digit_labels[i].config(fg=RED_DARK))
        except Exception:
            pass

    def force_top(self):
        try:
            self.root.lift()
            self.root.attributes("-topmost", True)
            self.root.focus_force()
        except Exception:
            pass

    def on_key(self, e):
        if self.deleting or self.unlocked:
            return "break"
        if e.char and e.char.isdigit():
            if len(self.password_buffer) < 4:
                self.password_buffer += e.char
                self.highlight(e.char)
                self.update_display()
                if len(self.password_buffer) == 4:
                    self.root.after(120, self.check)
            return "break"
        if e.keysym in ("Return", "KP_Enter"):
            self.check()
            return "break"
        if e.keysym == "BackSpace":
            self.password_buffer = self.password_buffer[:-1]
            self.update_display()
            return "break"
        return "break"

    def update_display(self):
        t = self.password_buffer
        z = "0" * (4 - len(t))
        self.display.config(text=t + z)

    def update_counter(self):
        self.counter.config(text="Осталось попыток: " +
                                 str(MAX_ATTEMPTS - self.attempts) +
                                 " / " + str(MAX_ATTEMPTS))

    def check(self):
        if self.deleting or self.unlocked:
            return
        if self.password_buffer == PASSWORD:
            self.unlocked = True
            self.status.config(text="ACCESS GRANTED", fg=GREEN)
            play_sound(SOUND_CORRECT)
            take_photo("correct")
            self.root.after(80, self.unlock)
        else:
            self.attempts += 1
            play_sound(SOUND_WRONG[min(self.attempts - 1, len(SOUND_WRONG) - 1)])
            take_photo("wrong_attempt" + str(self.attempts))
            self.password_buffer = ""
            self.update_display()
            self.update_counter()
            if self.attempts >= MAX_ATTEMPTS:
                self.start_deletion()
            else:
                self.status.config(text="НЕВЕРНЫЙ ПАРОЛЬ. Осталось: " +
                                       str(MAX_ATTEMPTS - self.attempts))

    def start_deletion(self):

        self.counter.config(text="УДАЛЕНИЕ ДАННЫХ...", fg=RED)
        self.warn.config(text="ФАЙЛЫ УДАЛЯЮТСЯ...", fg=RED)
        play_sound(SOUND_PANIC)
        self.log_frame.place(relx=0.5, rely=0.88, anchor="center")
        self.log_box.configure(state="normal")
        self.log_box.delete("1.0", tk.END)
        self.log_box.configure(state="disabled")
        threading.Thread(target=self.fake_delete_loop, daemon=True).start()

            self.log_box.configure(state="disabled")

            pass
